% ---------------------------------------------------------------- research
\cvsection{Research Experience}

\cventry{Ph.D. Research, Yale University}{2025 -- present}
  {Advisor: Prof.\ Yize Zhao. Deep generative models and statistical learning for multimodal
   neuroimaging, brain connectomics, and disease progression.}

\cventry{M.S. Research, University of North Carolina at Chapel Hill}{2023 -- 2025}{}
\vspace{-3pt}
\begin{itemize}
  \item \textit{Multimodal disentangled network VAE} (with Prof.\ Meimei Liu and Prof.\ Zhengwu Zhang), 2024 -- 2025.
    Graph variational autoencoders that separate shared and modality-specific components of
    structural and functional connectivity and predict traits from connectomes.
  \item \textit{Surface-based connectivity integration (SBCI) for the Human Connectome Project}
    (with Prof.\ Zhengwu Zhang), 2024 -- 2025.
    Preprocessed HCP-Aging and HCP-Development diffusion MRI, T1, and resting-state fMRI
    (FSL, MRtrix3) and extended the SBCI pipeline to generate connectivity under multiple atlases.
  \item \textit{EPIC: Prediction of Child Behavior Checklist scores}
    (with Prof.\ Guorong Wu and Prof.\ Zhengwu Zhang), Summer 2024.
    Combined connectome features (node2vec, JIVE) with XGBoost and Shapley-value feature
    selection to predict CBCL labels in the ABCD study.
  \item \textit{Structural connectomes for trait prediction in adolescents}
    (with Prof.\ Zhengwu Zhang), Spring 2024.
    Compared PCA-based SVM, logistic regression, and stacking models for predicting traits
    such as BMI from structural connectomes.
\end{itemize}
